// 누구를 클릭했는지에 따라 대화방 이름을 바꿔주는 함수
function enterChatRoom(name) {
    const chatRoom = document.getElementById('chat-room-area');
    const chatInput = document.getElementById('chat-input-area');
    const footer = document.querySelector('footer');

    // 1. 대화방 상단에 클릭한 사람 이름 넣기 (뒤로가기 버튼 옆에 이름 추가용)
    // 만약 대화방 안에 이름 넣는 곳이 없다면, 아래 로직으로 이름을 업데이트합니다.
    chatRoom.querySelector('.text-gray-400').innerText = name; 

    // 2. 화면 보이기/숨기기
    chatRoom.style.display = 'block';
    chatInput.style.display = 'block';
    if(footer) footer.style.display = 'none';
}


    // 2. 검색창 제어 (스윽 효과)
    if (searchTrigger && searchInput) {
        searchTrigger.onclick = (e) => {
            e.stopPropagation();
            isSearchOpen = !isSearchOpen;
            if (isSearchOpen) {
                if(logoHome) logoHome.classList.add('scale-0', 'w-0');
                if(navMenu) navMenu.classList.add('opacity-0', 'pointer-events-none');
                searchInput.classList.add('w-28', 'opacity-100', 'px-3');
                searchInput.focus();
            } else { closeSearch(); }
        };
    }

    function closeSearch() {
        isSearchOpen = false;
        if(logoHome) logoHome.classList.remove('scale-0', 'w-0');
        if(navMenu) navMenu.classList.remove('opacity-0', 'pointer-events-none');
        searchInput.classList.remove('w-28', 'opacity-100', 'px-3');
        searchInput.value = '';
    }

    // 3. 프로필 메뉴 및 외부 클릭 처리
    if (profileTrigger) {
        profileTrigger.onclick = () => {
            profileMenu.classList.add('open');
            menuOverlay.classList.remove('hidden');
        };
    }
    if (menuClose) menuClose.onclick = () => { profileMenu.classList.remove('open'); menuOverlay.classList.add('hidden'); };
    if (menuOverlay) menuOverlay.onclick = () => { profileMenu.classList.remove('open'); menuOverlay.classList.add('hidden'); };

    document.addEventListener('click', (e) => {
        if (isSearchOpen && !searchInput.contains(e.target) && e.target !== searchTrigger) closeSearch();
    });
