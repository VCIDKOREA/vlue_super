function enterChatRoom(userName) {
    let userData = null;
    for (const key in chatData) {
        const found = chatData[key].find(u => u.name === userName);
        if (found) { userData = found; break; }
    }
    if (!userData) return;

    showRoom();
    const nameSpan = document.getElementById('current-chat-user');
    if (nameSpan) nameSpan.innerText = userData.name;

    const messageContainer = document.getElementById('chat-content');
    if (messageContainer) {
        messageContainer.innerHTML = '';
        userData.chats.forEach(chat => {
            const isMe = chat.type === 'me';
            
            const bubble = isMe 
                ? `
                <div class="flex justify-end mb-4 px-4">
                    <div class="chat-bubble-unit bg-blue-600 text-white p-3 rounded-2xl rounded-tr-none max-w-[75%] w-fit text-[14px] shadow-sm">
                        ${chat.text}
                    </div>
                </div>`
                : `
                <div class="flex mb-4 px-4 items-start gap-2">
                    <div onclick="openProfileMiniMenu(event, '${userData.name}')" class="w-9 h-9 rounded-[13px] bg-gray-100 flex items-center justify-center shrink-0 cursor-pointer border border-gray-200">
                        ${defaultProfileIcon}

                    <div class="flex flex-col items-start">

                        </div>
                    </div>
                </div>`;
            messageContainer.insertAdjacentHTML('beforeend', bubble);
        });
        messageContainer.scrollTop = messageContainer.scrollHeight;
    }
}

    searchTrigger.onclick = (e) => {
        e.stopPropagation();
        isSearchOpen = !isSearchOpen;
        if (isSearchOpen) {
            if(logoHome) logoHome.classList.add('scale-0', 'w-0');
            if(navMenu) navMenu.classList.add('opacity-0', 'pointer-events-none');
            searchInput.classList.add('w-28', 'opacity-100', 'px-3');
            searchInput.focus();
        } else {
            closeSearch();
        }
    };
}

    !searchTrigger.contains(e.target) &&
    !searchInput.contains(e.target)) {
        if (isSearchOpen) closeSearch();
    }
});

    };

}

window.handleMenuAction = function(action) {
    const text = window.selectedChatText || "";
    const menu = document.getElementById('chat-context-menu');
    if (menu) menu.style.display = "none";

    if (action === 'ai') {
        setTimeout(() => {
            const result = `✨ [블루 AI 분석 리포트]\n\n💬 분석 메시지: "${text}"\n🧐 의도 파악: 질문 및 약속 확인\n💡 추천 답변: "6시쯤 모여요!", "장소는 어디인가요?"`;
            alert(result);
        }, 500);
    } else if (action === 'copy') {
        navigator.clipboard.writeText(text).then(() => alert("복사되었습니다."));
    } else if (action === 'translate') {
        alert("번역 기능을 준비 중입니다.");
    } else if (action === 'delete') {
        alert("삭제하시겠습니까?");
    }
};
function sendMessage() {
    const input = document.getElementById('chat-input');
    const content = document.getElementById('chat-content');

    if (!input.value.trim()) return;

    const bubble = `
    <div class="flex justify-end mb-2">
        <div class="chat-bubble-unit bg-blue-600 text-white p-3 rounded-2xl rounded-tr-none max-w-[80%] text-[14px]">
            ${input.value}
        </div>
    </div>`;

    content.insertAdjacentHTML('beforeend', bubble);
    input.value = "";
    content.scrollTop = content.scrollHeight;
}

            splash.style.opacity = '0';
            setTimeout(() => splash.style.display = 'none', 1000);
        }, 2000);
    }

    // 🔥 VCID 상태 연결
    const vcidToggle = document.getElementById('vcid-toggle');
    const saved = localStorage.getItem('vcid');

    // 저장값 반영
    if (saved === 'true') {
        isVCIDOn = true;
        if (vcidToggle) vcidToggle.checked = true;
    }

    // 이벤트
    if (vcidToggle) {
        vcidToggle.addEventListener('change', (e) => {
            isVCIDOn = e.target.checked;
            localStorage.setItem('vcid', isVCIDOn);

            console.log("VCID 상태:", isVCIDOn);

            // 🔥 여기부터 확장 가능
            if (isVCIDOn) {
                console.log("📞 VCID 활성화됨");
            } else {
                console.log("❌ VCID 비활성화됨");
            }
        });
    }

    // 기본 화면
    if (typeof showMain === 'function') {
        showMain();
    }
    if (!history.state) {
        window.history.pushState({ page: 'main' }, 'Main', window.location.href);
    }
});

function openExitModal() {
    const modal = document.getElementById('exit-modal');
    if(modal) modal.classList.remove('hidden');
}

// 팝업 닫기(취소) 함수
function closeExitModal() {
    const modal = document.getElementById('exit-modal');
    if(modal) modal.classList.add('hidden');
    // 취소했으니 다시 기록을 심어서 다음 뒤로가기에 대비
    window.history.pushState({ page: 'main' }, 'Main', window.location.href);
}

    window.close();
    // 브라우저에서 안 닫힐 경우 대비용
    setTimeout(() => { history.back(); }, 100);
}

window.onpopstate = function(event) {
    // 1. 채팅방이 열려있으면 목록으로
    if (chatRoomArea && chatRoomArea.style.display === 'flex') {
        exitChat();
    } 
    // 2. 프로필 메뉴가 열려있으면 닫기
    else if (profileMenu && profileMenu.classList.contains('open')) {
        menuClose.onclick();
    }
    // 3. 채팅 목록이 열려있으면 메인으로
    else if (chatListArea && chatListArea.style.display === 'block') {
        goMain();
    }
    // 4. 메인 화면에서 뒤로가기를 누르면 비로소 종료 팝업!
    else {
        openExitModal();
        // 팝업이 떴을 때 뒤로가기를 한 번 더 누르면 진짜 나갈 수 있도록 기록 유지
        window.history.pushState({ page: 'main' }, 'Main', window.location.href);
    }
};
